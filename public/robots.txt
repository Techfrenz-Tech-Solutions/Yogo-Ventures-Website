# ── Default: Allow all well-behaved crawlers ───────────────────────────────────

# ── Explicitly allow key indexable paths (reinforces priority for crawlers

Allow: /services
Allow: /contact

# ── Disallow transactional & user-only pages (no SEO value) ───────────────────
# NOTE: specific disallows must appear AFTER specific allows for longest-match
#       parsers to correctly prefer the Allow: /shop?category= rule above.
Disallow: /checkout
Disallow: /wishlist

# ── Disallow internal API, admin, and private paths ──────────────────────────
Disallow: /api/
Disallow: /admin/
Disallow: /private/

# ── Disallow duplicate-content query strings ─────────────────────────────────

# They are lower-specificity than the Allow: /shop?category= rule above,
# so category filter pages remain crawlable.
Disallow: /shop?sort=
Disallow: /shop?q=
